cmake_minimum_required(VERSION 3.16)
project(rmlui_construction_probes LANGUAGES CXX)

if(NOT DEFINED RMLUI_SOURCE)
  message(FATAL_ERROR "RMLUI_SOURCE is required")
endif()
if(NOT DEFINED PROBE_ASSETS)
  message(FATAL_ERROR "PROBE_ASSETS is required")
endif()

set(PROBE_GROUP core CACHE STRING "Isolated probe family: core, sdl, gl")

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
if(PROBE_GROUP STREQUAL "core")
  set(RMLUI_FONT_ENGINE freetype CACHE STRING "" FORCE)
else()
  set(RMLUI_FONT_ENGINE none CACHE STRING "" FORCE)
endif()
set(RMLUI_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
set(RMLUI_COMPILER_OPTIONS OFF CACHE BOOL "" FORCE)

if(PROBE_GROUP STREQUAL "gl")
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
add_subdirectory("${RMLUI_SOURCE}" rmlui EXCLUDE_FROM_ALL)

if(PROBE_GROUP STREQUAL "core")
  add_executable(node_probe "${PROBE_ASSETS}/probe.cpp")
  target_compile_features(node_probe PRIVATE cxx_std_17)
  target_link_libraries(node_probe PRIVATE rmlui_core)
  add_executable(render_paths_probe "${PROBE_ASSETS}/render_paths_probe.cpp")
  target_compile_features(render_paths_probe PRIVATE cxx_std_17)
  target_link_libraries(render_paths_probe PRIVATE rmlui_core rmlui_debugger)
endif()

if(PROBE_GROUP STREQUAL "sdl")
  find_package(SDL3 CONFIG REQUIRED)

add_executable(sdl_density_probe
  "${PROBE_ASSETS}/sdl_density_probe.cpp"
  "${RMLUI_SOURCE}/Backends/RmlUi_Platform_SDL.cpp"
)
target_compile_features(sdl_density_probe PRIVATE cxx_std_17)
target_compile_definitions(sdl_density_probe PRIVATE RMLUI_SDL_VERSION_MAJOR=3 PROBE_SDL_MAJOR=3)
target_include_directories(sdl_density_probe PRIVATE "${RMLUI_SOURCE}/Backends")
target_link_libraries(sdl_density_probe PRIVATE rmlui_core SDL3::SDL3)
  target_link_options(sdl_density_probe PRIVATE "-Wl,--wrap=SDL_GetWindowDisplayScale")
endif()

if(PROBE_GROUP STREQUAL "gl")
find_package(OpenGL REQUIRED)
add_executable(gl_texture_guard_probe
  "${PROBE_ASSETS}/gl_texture_guard_probe.cpp"
  "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_GL2.cpp"
)
target_compile_features(gl_texture_guard_probe PRIVATE cxx_std_17)
target_compile_definitions(gl_texture_guard_probe PRIVATE RMLUI_DEBUG)
target_include_directories(gl_texture_guard_probe PRIVATE "${RMLUI_SOURCE}/Backends")
target_link_libraries(gl_texture_guard_probe PRIVATE rmlui_core OpenGL::GL)
target_link_options(gl_texture_guard_probe PRIVATE "-Wl,--wrap=glGenTextures" "-Wl,--wrap=glBindTexture" "-Wl,--wrap=glTexImage2D" "-Wl,--wrap=glTexParameteri")
endif()

if(PROBE_GROUP STREQUAL "gl")
  find_package(SDL2 REQUIRED)
  find_package(SDL2_image REQUIRED)
  find_package(SDL3 CONFIG REQUIRED)
  find_package(SDL3_image CONFIG REQUIRED PATHS /opt/sdl3-image)
  add_executable(texture_sdl3_probe "${PROBE_ASSETS}/backend_texture_probe.cpp" "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_SDL.cpp")
  target_compile_features(texture_sdl3_probe PRIVATE cxx_std_17)
  target_compile_definitions(texture_sdl3_probe PRIVATE RMLUI_DEBUG PROBE_SDL RMLUI_SDL_VERSION_MAJOR=3)
  target_include_directories(texture_sdl3_probe PRIVATE "${RMLUI_SOURCE}/Backends")
  target_link_libraries(texture_sdl3_probe PRIVATE rmlui_core SDL3::SDL3 SDL3_image::SDL3_image)
  foreach(backend GL3 SDL)
    string(TOLOWER "${backend}" lower)
    add_executable(texture_${lower}_probe "${PROBE_ASSETS}/backend_texture_probe.cpp" "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_${backend}.cpp")
    target_compile_features(texture_${lower}_probe PRIVATE cxx_std_17)
    target_compile_definitions(texture_${lower}_probe PRIVATE RMLUI_DEBUG PROBE_${backend})
    target_include_directories(texture_${lower}_probe PRIVATE "${RMLUI_SOURCE}/Backends")
    target_link_libraries(texture_${lower}_probe PRIVATE rmlui_core)
  endforeach()
  target_compile_definitions(texture_gl3_probe PRIVATE RMLUI_GL3_CUSTOM_LOADER="GL/glcorearb.h" GL_GLEXT_PROTOTYPES)
  target_link_libraries(texture_gl3_probe PRIVATE OpenGL::GL)
  foreach(symbol glGenTextures glBindTexture glTexImage2D glTexParameteri glCreateShader glCreateProgram glGetShaderiv glGetProgramiv glGenFramebuffers glGenRenderbuffers glCheckFramebufferStatus)
    target_link_options(texture_gl3_probe PRIVATE "-Wl,--wrap=${symbol}")
  endforeach()
  target_link_libraries(texture_sdl_probe PRIVATE SDL2::SDL2 SDL2_image::SDL2_image)
  target_compile_definitions(texture_sdl_probe PRIVATE RMLUI_SDL_VERSION_MAJOR=2)
  add_library(texture_vk_renderer SHARED "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_VK.cpp")
  target_compile_features(texture_vk_renderer PRIVATE cxx_std_17)
  target_compile_definitions(texture_vk_renderer PRIVATE RMLUI_DEBUG)
  target_include_directories(texture_vk_renderer PUBLIC "${RMLUI_SOURCE}/Backends")
  target_link_libraries(texture_vk_renderer PUBLIC rmlui_core PRIVATE dl)
  add_executable(texture_vk_probe "${PROBE_ASSETS}/vulkan_texture_guard_probe.cpp")
  target_compile_features(texture_vk_probe PRIVATE cxx_std_17)
  target_link_libraries(texture_vk_probe PRIVATE texture_vk_renderer)
  target_link_options(texture_vk_probe PRIVATE "-Wl,--export-dynamic-symbol=vmaCreateBuffer")
endif()
